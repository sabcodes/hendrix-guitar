cmake_minimum_required(VERSION 3.22)
project(HendrixGuitar VERSION 0.2.0 LANGUAGES C CXX)
add_subdirectory(JUCE)
juce_add_plugin(HendrixGuitar COMPANY_NAME sabcodes BUNDLE_ID com.sabcodes.hendrixguitar PLUGIN_MANUFACTURER_CODE Hdxg PLUGIN_CODE Hdg1 FORMATS AU IS_SYNTH TRUE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT FALSE PRODUCT_NAME "Hendrix Guitar" COPY_PLUGIN_AFTER_BUILD FALSE)
target_sources(HendrixGuitar PRIVATE Source/GuitarEngine.cpp Source/VintageTone.cpp Source/Parameters.cpp Source/PluginProcessor.cpp Source/PluginEditor.cpp)
target_include_directories(HendrixGuitar PUBLIC Source)
target_compile_features(HendrixGuitar PUBLIC cxx_std_17)
target_compile_definitions(HendrixGuitar PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(HendrixGuitar PUBLIC juce::juce_audio_utils juce::juce_dsp PRIVATE juce::juce_recommended_config_flags)
set_target_properties(HendrixGuitar_AU PROPERTIES LIBRARY_OUTPUT_DIRECTORY "$ENV{HOME}/Library/Caches/HendrixGuitar/AU")
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/au-bundle-path.txt" CONTENT "$<TARGET_BUNDLE_DIR:HendrixGuitar_AU>")
add_custom_command(TARGET HendrixGuitar_AU POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/Resources/Guitar" "$<TARGET_BUNDLE_DIR:HendrixGuitar_AU>/Contents/Resources/Guitar" COMMAND /usr/bin/xattr -cr "$<TARGET_BUNDLE_DIR:HendrixGuitar_AU>" COMMAND /usr/bin/codesign --force --sign - "$<TARGET_BUNDLE_DIR:HendrixGuitar_AU>")
include(CTest)
if(BUILD_TESTING)
 find_package(Python3 REQUIRED COMPONENTS Interpreter)
 add_executable(HendrixGuitarChecks)
 target_sources(HendrixGuitarChecks PRIVATE Tests/TestMain.cpp Tests/GuitarChecks.cpp Tests/StateChecks.cpp)
 target_link_libraries(HendrixGuitarChecks PRIVATE HendrixGuitar)
 add_test(NAME GuitarChecks COMMAND HendrixGuitarChecks)
 add_test(NAME BundleChecks COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/Tests/BundleChecks.py "$<TARGET_BUNDLE_DIR:HendrixGuitar_AU>")
 add_test(NAME SampleImportChecks COMMAND ${Python3_EXECUTABLE} -m unittest discover -s ${CMAKE_CURRENT_SOURCE_DIR}/Tests -p SampleImportChecks.py -v)
endif()
